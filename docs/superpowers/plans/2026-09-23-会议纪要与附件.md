不存在 jest / vitest，也没有「只跑某个用例」的开关。**自动化验证只有 4 条**，每条都是全量跑：

| 命令 | 内容 | 前置 |
|------|------|------|
| `npm run typecheck` | `tsc --noEmit` | 无 |
| `npm run build` | 生产构建 | **必须先停服** |
| `npm run api:smoke` | 接口冒烟，**119 项**，本计划会扩到 ~140 | **服务须在 :3000 运行** |
| `npm run check:sort` | 排序断言 15 项 | 无 |

「写测试」在本项目里 = **往 `scripts/api-smoke.mjs` 里加断言**。它自清理、可重复运行。

### 2. Windows：服务运行时会锁住 `.next`

构建报 EPERM 就是它。跑 `npm run build` 前必须停服，且要连子进程一起清：

```bash
netstat -ano | grep LISTENING | grep ":3000"          # 拿 PID
taskkill //F //PID <PID>
```

### 3. ⚠️ 命令行里的中文会被控制台按 GBK 处理

`curl -d '{"title":"周会"}'` 发出去的中文**到达服务端时已经损坏**（字节里出现 `EFBFBD`）。
用中文做接口验证会得到**假故障**。

- 能用 ASCII 就用 ASCII（如系统字段 `OMS`）
- 必须中文时：用 Python（源码写 `\uXXXX` 转义，纯 ASCII）生成 UTF-8 的 body 文件，再 `curl -d @file`
- 排查时**看 `hex()` 输出，不要看终端回显** —— 终端显示的中文同样不可信

### 4. 子会话分工（本次按子会话驱动规范执行）

无头子会话拿不到交互式 Bash 审批。因此：

- **子会话只产出文件**（Read/Write/Edit/Glob/Grep）
- **迁移、typecheck、build、冒烟、curl 由总调度执行**
- **退出码 0 ≠ 验收通过** —— 总调度必须独立读文件 + 跑运行级命令

本计划中凡标注 **【总调度执行】** 的步骤，不下发给子会话。

```ts
/* ---------------- 会议纪要 ---------------- */

export interface MeetingAttachmentJson {
  id: string;
  fileName: string;
  size: number;
  mimeType?: string;
  createdAt: string;
}

export interface MeetingJson {
  id: string;
  date: string;
  title: string;
  attendees: string[];
  recorder?: string;
  content?: string;
  attachments: MeetingAttachmentJson[];
  createdAt: string;
  updatedAt: string;
}

/**
 * 解析库里的 attendees（JSON 数组文本）。
 * ⚠️ 脏数据（非法 JSON / 不是数组 / 元素不是字符串）一律回退成空数组，
 * **绝不抛错** —— 一条坏数据不能让整个列表页 500。
 */
export function parseAttendees(raw: string | null | undefined): string[] {
  if (!raw) return [];
  try {
    const v: unknown = JSON.parse(raw);
    if (!Array.isArray(v)) return [];
    return v
      .filter((x): x is string => typeof x === "string")
      .map((s) => s.trim())
      .filter(Boolean);
  } catch {
    return [];
  }
}

/**
 * 参会人数组 → 库里的 JSON 文本。
 * undefined（未传）→ undefined（不改该字段）；[] 或全空白 → null（清空）。
 */
export function serializeAttendees(list: unknown): string | null | undefined {
  if (list === undefined) return undefined;
  if (!Array.isArray(list)) return null;
  const arr = list
    .filter((x): x is string => typeof x === "string")
    .map((s) => s.trim())
    .filter(Boolean);
  return arr.length ? JSON.stringify(arr) : null;
}

export function toMeetingAttachmentJson(
  a: DbMeetingAttachment,
): MeetingAttachmentJson {
  return {
    id: a.id,
    fileName: a.fileName,
    size: a.size,
    mimeType: clean(a.mimeType),
    createdAt: a.createdAt.toISOString(),
  };
}

export function toMeetingJson(
  m: DbMeeting & { attachments?: DbMeetingAttachment[] },
): MeetingJson {
  return {
    id: m.id,
    date: dayOnly(m.date) as string,
    title: m.title,
    attendees: parseAttendees(m.attendees),
    recorder: clean(m.recorder),
    content: clean(m.content),
    attachments: (m.attachments ?? []).map(toMeetingAttachmentJson),
    createdAt: m.createdAt.toISOString(),
    updatedAt: m.updatedAt.toISOString(),
  };
}
```

```ts
import { NextResponse } from "next/server";
import type { Prisma } from "@prisma/client";
import { prisma } from "@/lib/server/db";
import {
  apiError,
  asTrimmed,
  isValidCalendarDay,
  readJson,
  serializeAttendees,
  toDayUtc,
  toMeetingJson,
} from "@/lib/server/helpers";

export const runtime = "nodejs";

/**
 * GET /api/meetings
 * 会议纪要列表。可选过滤：?keyword=xxx（匹配 主题 / 记录人 / 会议内容 / 参会人）
 * ?attendee=张三（按人筛，同样是子串匹配，见设计文档「明确不做」）。
 * 按创建时间倒序，含附件列表。
 */
export async function GET(req: Request) {
  const url = new URL(req.url);
  const keyword = url.searchParams.get("keyword")?.trim();
  const attendee = url.searchParams.get("attendee")?.trim();

  const where: Prisma.MeetingWhereInput = {
    ...(keyword
      ? {
          OR: [
            { title: { contains: keyword } },
            { recorder: { contains: keyword } },
            { content: { contains: keyword } },
            { attendees: { contains: keyword } },
          ],
        }
      : {}),
    ...(attendee ? { attendees: { contains: attendee } } : {}),
  };

  const rows = await prisma.meeting.findMany({
    where,
    orderBy: { createdAt: "desc" },
    include: { attachments: true },
  });
  return NextResponse.json(rows.map(toMeetingJson));
}

/**
 * POST /api/meetings
 * 新建会议纪要（**纯 JSON，不接收文件** —— 附件由 POST /api/meetings/{id}/attachments 上传）。
 * body：date*（YYYY-MM-DD）/ title* 必填；attendees（字符串数组）/ recorder / content 可选。
 * 返回 201 + 创建结果。
 */
export async function POST(req: Request) {
  const body = await readJson(req);
  if (!body) return apiError(400, "请求体必须是合法 JSON 对象。");

  const errors: string[] = [];

  const date = asTrimmed(body.date);
  if (!date || !isValidCalendarDay(date)) {
    errors.push("date 必填，格式 YYYY-MM-DD。");
  }

  const title = asTrimmed(body.title);
  if (!title) errors.push("title（主题）必填。");

  // attendees 必须是字符串数组（或缺省）；传其它类型直接 400，不要静默吞掉
  if (body.attendees !== undefined && !Array.isArray(body.attendees)) {
    errors.push("attendees 须为字符串数组。");
  }

  if (errors.length) return apiError(400, "参数校验失败。", errors);

  const attendees = serializeAttendees(body.attendees);
  const recorder = asTrimmed(body.recorder);
  const content = asTrimmed(body.content);

  const created = await prisma.meeting.create({
    data: {
      date: toDayUtc(date as string) as Date,
      title: title as string,
      ...(attendees ? { attendees } : {}),
      ...(recorder ? { recorder } : {}),
      ...(content ? { content } : {}),
    },
    include: { attachments: true },
  });
  return NextResponse.json(toMeetingJson(created), { status: 201 });
}
```

```ts
/** GET /api/meetings/{id} —— 单条（含附件） */
export async function GET(_req: Request, { params }: Ctx) {
  const row = await prisma.meeting.findUnique({
    where: { id: params.id },
    include: { attachments: true },
  });
  if (!row) return apiError(404, "会议纪要不存在。");
  return NextResponse.json(toMeetingJson(row));
}

/**
 * PUT /api/meetings/{id}
 * 三态语义（与需求模块一致，不得自创）：
 *   undefined（未传）→ 不改该字段；""（显式清空）→ 落 null；非空 → 改。
 * attendees 额外约定：传数组则整体替换；传 [] 视为清空；未传则不动。**不做数组级增量合并**。
 */
export async function PUT(req: Request, { params }: Ctx) {
  const body = await readJson(req);
  if (!body) return apiError(400, "请求体必须是合法 JSON 对象。");

  const existing = await prisma.meeting.findUnique({ where: { id: params.id } });
  if (!existing) return apiError(404, "会议纪要不存在。");

  const errors: string[] = [];
  const data: Prisma.MeetingUpdateInput = {};

  if (body.date !== undefined) {
    const date = asTrimmed(body.date);
    if (!date || !isValidCalendarDay(date)) {
      errors.push("date 格式须为 YYYY-MM-DD。");
    } else {
      data.date = toDayUtc(date) as Date;
    }
  }

  if (body.title !== undefined) {
    const title = asTrimmed(body.title);
    if (!title) errors.push("title（主题）不能为空。");
    else data.title = title;
  }

  if (body.attendees !== undefined) {
    if (!Array.isArray(body.attendees)) {
      errors.push("attendees 须为字符串数组。");
    } else {
      data.attendees = serializeAttendees(body.attendees) ?? null;
    }
  }

  if (body.recorder !== undefined) data.recorder = asTrimmed(body.recorder) || null;
  if (body.content !== undefined) data.content = asTrimmed(body.content) || null;

  if (errors.length) return apiError(400, "参数校验失败。", errors);

  const updated = await prisma.meeting.update({
    where: { id: params.id },
    data,
    include: { attachments: true },
  });
  return NextResponse.json(toMeetingJson(updated));
}

/**
 * DELETE /api/meetings/{id}
 * 1) 先查出所有附件的 storedName
 * 2) 删纪要（附件记录随 onDelete: Cascade 清掉）
 * 3) **事务提交之后**再逐个删磁盘文件；删文件失败只记日志，不影响接口返回
 *
 * 顺序不能颠倒：先删文件后删记录，一旦第二步失败就会留下「记录还在、文件没了」
 * 的不可自愈状态；反过来最坏只留一个孤儿文件。
 */
export async function DELETE(_req: Request, { params }: Ctx) {
  const existing = await prisma.meeting.findUnique({
    where: { id: params.id },
    include: { attachments: true },
  });
  if (!existing) return apiError(404, "会议纪要不存在。");

  const storedNames = existing.attachments.map((a) => a.storedName);

  await prisma.meeting.delete({ where: { id: params.id } });

  // 记录已删，现在清理磁盘。失败只记日志 —— 孤儿文件人畜无害，
  // 而「记录还在文件没了」是不可自愈的。
  for (const name of storedNames) {
    await removeStored(name);
  }

  return new NextResponse(null, { status: 204 });
}
```

```tsx
 * 候选人名由调用方传入（本项目的来源：已加载的全部会议纪要里出现过的

  onChange?: (v: string | string[] | undefined) => void;
  /** 候选人名 */
  options: string[];

  placeholder?: string;
  style?: React.CSSProperties;

  value,
  onChange,
  options,
  multiple = false,
  placeholder,
  style,
}: PersonSelectProps) {
  const selectOptions = useMemo(
    () => options.map((name) => ({ label: name, value: name })),
    [options],
  );

  return (
    <Select
      // mode="tags" 才允许输入候选之外的新值；"multiple" 只能选已有项

      // 单选场景下 tags 也能用，但要去掉多值行为

      showSearch

  );
}
```

```ts
import { randomUUID } from "node:crypto";
import { mkdir, unlink, writeFile } from "node:fs/promises";
import path from "node:path";
import { ApiFailure } from "@/lib/server/helpers";

/**
 * 附件的磁盘读写、命名与校验 —— 本项目**唯一**的文件 IO 实现处。
 * 上传 / 下载 / 删除三条路径共用本文件，规则不得在别处重写
 * （与 work-order-sync.ts 之于「工单转需求」同理）。
 *
 * ⚠️ 安全前提：file.originalname 是**用户可控输入**，可能是 `../../.env`、
 * `截图 2026/09/23.png`、Windows 保留名（CON/NUL）、超长名或带 \0。
 * 因此：
 *   - 磁盘文件名一律由服务端生成（UUID），**不含任何用户输入**
 *   - 原始名只写进数据库，用于展示与下载时的文件名
 *   - 扩展名必须命中白名单才放行
 */

export const UPLOAD_DIR = path.join(process.cwd(), "uploads");

/** 单文件上限 20MB（设计文档约定） */
export const MAX_FILE_BYTES = 20 * 1024 * 1024;

/** 允许的扩展名（小写，含点） */
export const ALLOWED_EXT: ReadonlySet<string> = new Set([
  // 图片
  ".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp",
  // 文档
  ".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx",
  // 文本
  ".txt", ".csv",
  // 压缩包
  ".zip", ".rar", ".7z",
]);

/** 供错误提示用的人类可读列表 */
export const ALLOWED_EXT_TEXT = Array.from(ALLOWED_EXT).join(" / ");

/**
 * 从原始文件名取扩展名并校验白名单。
 * 返回 null 表示不允许（无扩展名、隐藏文件、或不在白名单）。
 */
export function safeExt(originalName: string): string | null {
  // 先取 basename：浏览器偶尔会带完整路径；同时挡掉 `../../x.txt` 这类
  const base = path.basename(originalName || "");
  const i = base.lastIndexOf(".");
  // i <= 0 同时排除「无扩展名」和「.env 这类隐藏文件」
  if (i <= 0) return null;
  const ext = base.slice(i).toLowerCase();
  return ALLOWED_EXT.has(ext) ? ext : null;
}

/**
 * 把库里的 storedName 还原成磁盘绝对路径，并**断言不越界**。
 * storedName 由服务端生成，理论上不会越界；这道断言是纵深防御 ——
 * 万一将来有人往库里写了脏值，也不要让它读到 uploads/ 之外。
 */
export function resolveStored(storedName: string): string {
  if (!storedName || storedName.includes("/") || storedName.includes("\\")) {
    throw new Error(`非法的 storedName：${storedName}`);
  }
  const full = path.join(UPLOAD_DIR, storedName);
  if (path.dirname(full) !== UPLOAD_DIR) {
    throw new Error("附件路径越界");
  }
  return full;
}

export interface SavedUpload {
  fileName: string;   // 原始文件名（写库用）
  storedName: string; // 磁盘文件名（UUID + 白名单后缀）
  size: number;
  mimeType: string | null;
}

/**
 * 校验并落盘。失败抛 ApiFailure（由路由 catch 转成标准错误响应）。
 * 任一步失败都不留下半截文件。
 */
export async function saveUpload(file: File): Promise<SavedUpload> {
  const ext = safeExt(file.name);
  if (!ext) {
    throw new ApiFailure(400, `不支持的文件类型，允许：${ALLOWED_EXT_TEXT}`);
  }
  if (file.size > MAX_FILE_BYTES) {
    throw new ApiFailure(400, "单个附件不能超过 20MB。");
  }

  const storedName = `${randomUUID()}${ext}`;
  const buf = Buffer.from(await file.arrayBuffer());

  // 缓冲后 size 可能比 file.size 更可信，再卡一次
  if (buf.length > MAX_FILE_BYTES) {
    throw new ApiFailure(400, "单个附件不能超过 20MB。");
  }

  await mkdir(UPLOAD_DIR, { recursive: true });
  await writeFile(resolveStored(storedName), buf);

  return {
    fileName: path.basename(file.name || "未命名"),
    storedName,
    size: buf.length,
    // ⚠️ mimeType 来自客户端，可伪造，**只存作参考，绝不用于安全判断**
    mimeType: file.type || null,
  };
}

/**
 * 删除磁盘文件。
 * **失败只记日志、不抛错** —— 调用方是在数据库记录已删除之后执行的，
 * 此时抛错只会让用户以为删除失败，而实际记录已经没了。
 * 最坏结果是留一个孤儿文件：它不占多少空间，也不会被任何界面引用。
 */
export async function removeStored(storedName: string): Promise<void> {
  try {
    await unlink(resolveStored(storedName));
  } catch (e) {
    const code = (e as NodeJS.ErrnoException).code;
    if (code === "ENOENT") return; // 本来就不在，视为成功
    console.error("[attachments] 删除附件文件失败：", storedName, code);
  }
}
```

```ts
import { NextResponse } from "next/server";
import { prisma } from "@/lib/server/db";
import { ApiFailure, apiError, toMeetingAttachmentJson } from "@/lib/server/helpers";
import { MAX_FILE_BYTES, removeStored, saveUpload } from "@/lib/server/attachments";

export const runtime = "nodejs";

interface Ctx {
  params: { id: string };
}

/**
 * POST /api/meetings/{id}/attachments
 * multipart/form-data，字段名 `file`。返回 201 + 附件记录。
 *
 * 两道大小关：
 *   ① Content-Length 预检 —— 超限直接 413，**在读取 body 之前**就拒绝
 *   ② 实际 file.size 超限 → 400（由 saveUpload 抛 ApiFailure）
 *
 * ⚠️ 这不是流式校验：request.formData() 会把整个 body 缓冲完才交给代码。
 * 单用户内网场景可接受；若要开放给更多人，须改为流式 multipart 解析。
 */
export async function POST(req: Request, { params }: Ctx) {
  const meeting = await prisma.meeting.findUnique({ where: { id: params.id } });
  if (!meeting) return apiError(404, "会议纪要不存在。");

  const len = Number(req.headers.get("content-length") || 0);
  // +1MB 余量给 multipart 的边界与头
  if (len > MAX_FILE_BYTES + 1024 * 1024) {
    return apiError(413, "请求体过大，单个附件不能超过 20MB。");
  }

  let form: FormData;
  try {
    form = await req.formData();
  } catch {
    return apiError(400, "请求必须是 multipart/form-data。");
  }

  const file = form.get("file");
  if (!(file instanceof File)) {
    return apiError(400, "缺少文件字段 file。");
  }

  let saved;
  try {
    saved = await saveUpload(file);
  } catch (e) {
    if (e instanceof ApiFailure) return apiError(e.status, e.message);
    throw e;
  }

  try {
    const row = await prisma.meetingAttachment.create({
      data: {
        meetingId: params.id,
        fileName: saved.fileName,
        storedName: saved.storedName,
        size: saved.size,
        mimeType: saved.mimeType,
      },
    });
    return NextResponse.json(toMeetingAttachmentJson(row), { status: 201 });
  } catch (e) {
    // 写库失败 → 把刚落盘的文件删掉，不留垃圾
    await removeStored(saved.storedName);
    throw e;
  }
}
```

```ts
import { NextResponse } from "next/server";
import { createReadStream } from "node:fs";
import { stat } from "node:fs/promises";
import { Readable } from "node:stream";
import { prisma } from "@/lib/server/db";
import { apiError } from "@/lib/server/helpers";
import { removeStored, resolveStored } from "@/lib/server/attachments";

export const runtime = "nodejs";

interface Ctx {
  params: { id: string };
}

/** 图片直接内联显示（页面里 <img src> 就能用），其余触发下载 */
const INLINE_EXT = new Set([".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp", ".pdf"]);

/**
 * GET /api/attachments/{id} —— 下载 / 预览
 * - 记录不存在 → 404
 * - 记录在但文件不在 → 404 + 明确文案（**不要 500**，也不静默返回空文件）
 */
export async function GET(_req: Request, { params }: Ctx) {
  const row = await prisma.meetingAttachment.findUnique({ where: { id: params.id } });
  if (!row) return apiError(404, "附件不存在。");

  let full: string;
  try {
    full = resolveStored(row.storedName);
  } catch {
    return apiError(404, "附件的存储路径非法。");
  }

  let size: number;
  try {
    size = (await stat(full)).size;
  } catch {
    return apiError(404, "附件文件已丢失（可能被手工删除），请联系管理员。");
  }

  const ext = row.storedName.slice(row.storedName.lastIndexOf(".")).toLowerCase();
  const disposition = INLINE_EXT.has(ext) ? "inline" : "attachment";

  // ⚠️ 中文文件名必须双写：只给 filename= 的话浏览器会乱码或截断
  const utf8Name = encodeURIComponent(row.fileName);
  const asciiFallback = row.fileName.replace(/[^\x20-\x7E]/g, "_").replace(/"/g, "");

  const stream = Readable.toWeb(createReadStream(full)) as ReadableStream;
  return new NextResponse(stream, {
    status: 200,
    headers: {
      "Content-Type": row.mimeType || "application/octet-stream",
      "Content-Length": String(size),
      "Content-Disposition":
        `${disposition}; filename="${asciiFallback}"; filename*=UTF-8''${utf8Name}`,
      "Cache-Control": "private, max-age=0, must-revalidate",
    },
  });
}

/** DELETE /api/attachments/{id} —— 先删记录，再删文件（理由见设计文档 3.3） */
export async function DELETE(_req: Request, { params }: Ctx) {
  const row = await prisma.meetingAttachment.findUnique({ where: { id: params.id } });
  if (!row) return apiError(404, "附件不存在。");

  await prisma.meetingAttachment.delete({ where: { id: params.id } });
  await removeStored(row.storedName);

  return new NextResponse(null, { status: 204 });
}
```

```tsx
"use client";

import { useCallback, useState } from "react";
import { useRouter } from "next/navigation";
import { message } from "antd";
import { useAppStore } from "@/store/store";
import { meetingApi } from "@/lib/api";
import { toErrorMessage } from "@/lib/errors";
import MeetingForm, { type MeetingFormValues } from "@/components/MeetingForm";

export default function CreateMeetingPage() {
  const router = useRouter();
  const addMeeting = useAppStore((s) => s.addMeeting);
  const addMeetingAttachment = useAppStore((s) => s.addMeetingAttachment);
  const [submitting, setSubmitting] = useState(false);
  const [pendingFiles, setPendingFiles] = useState<File[]>([]);

  // 用 useCallback 固定引用，否则 MeetingForm 里的 useEffect 会反复触发
  const handlePendingFilesChange = useCallback((files: File[]) => {
    setPendingFiles(files);
  }, []);

  const handleSubmit = async (values: MeetingFormValues) => {
    setSubmitting(true);
    try {
      // ① 先把纪要建出来，拿到 id
      const created = await addMeeting({
        date: values.date.format("YYYY-MM-DD"),
        title: values.title,
        attendees: values.attendees ?? [],
        recorder: values.recorder || undefined,
        content: values.content || undefined,
      });

      // ② 再逐个传附件
      const failed: string[] = [];
      for (const f of pendingFiles) {
        try {
          const att = await meetingApi.uploadAttachment(created.id, f);
          addMeetingAttachment(created.id, att);
        } catch {
          failed.push(f.name);
        }
      }

      // ③ 如实汇报 —— 不得把「部分失败」说成成功
      if (failed.length) {
        message.warning(
          `会议纪要已保存，但 ${failed.length} 个附件上传失败：${failed.join("、")}。请在编辑页重新上传。`,
        );
      } else if (pendingFiles.length) {
        message.success(`会议纪要已创建，${pendingFiles.length} 个附件已上传`);
      } else {
        message.success("会议纪要已创建");
      }

      // ④ 一律跳编辑页：上传失败的附件能立刻在这儿重传
      router.push(`/meetings/edit/${created.id}`);
    } catch (e) {
      message.error(toErrorMessage(e));
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <MeetingForm
      submitting={submitting}
      onSubmit={handleSubmit}
      onCancel={() => router.push("/meetings")}
      onPendingFilesChange={handlePendingFilesChange}
    />
  );
}
```

```js
const backupDir = resolve(root, "backups");
if (!existsSync(backupDir)) mkdirSync(backupDir, { recursive: true });

// 附件目录（会议纪要上传的文件）。可能不存在 —— 尚无任何附件时就是没有
const uploadsDir = resolve(root, "uploads");
const hasUploads = existsSync(uploadsDir);

// 时间戳文件名
const now = new Date();
const pad = (n, w = 2) => String(n).padStart(w, "0");
const stamp =
  `${now.getFullYear()}${pad(now.getMonth() + 1)}${pad(now.getDate())}` +
  `-${pad(now.getHours())}${pad(now.getMinutes())}${pad(now.getSeconds())}`;

// 产物：db + uploads 一起打包
const target = resolve(backupDir, `${stamp}.tar.gz`);
// VACUUM INTO 要求目标文件不存在，故快照先落在暂存目录
const stageDir = resolve(backupDir, `.staging-${stamp}`);
mkdirSync(stageDir, { recursive: true });
const snapshot = resolve(stageDir, "dev.db");

process.env.DATABASE_URL = dbUrl;
const { PrismaClient } = require("@prisma/client");
const prisma = new PrismaClient();

try {
  // 记录备份前的数据量，便于恢复后核对
  const [wo, req, att] = await Promise.all([
    prisma.workOrder.count(),
    prisma.requirement.count(),
    prisma.meetingAttachment.count(),
  ]);

  // SQLite 路径在 SQL 里用正斜杠，避免反斜杠转义问题
  const sqlPath = snapshot.replace(/\\/g, "/").replace(/'/g, "''");
  await prisma.$executeRawUnsafe(`VACUUM INTO '${sqlPath}'`);

  // 打成 tar.gz。用 execFileSync（数组参数）而非 shell 拼串 ——
  // 本项目路径含空格（`四川物流工单系统`），走 shell 会被拆成两个参数。
  //
  // tar 的 -C 对**其后**的文件参数生效，所以可以分段指定来源目录：
  // 先取暂存目录里的 dev.db，再取仓库根下的 uploads/。
  // 这样**不必把 uploads 复制一份**，大附件不会让磁盘占用翻倍。
  const tarArgs = ["-czf", target, "-C", stageDir, "dev.db"];
  if (hasUploads) tarArgs.push("-C", root, "uploads");
  execFileSync("tar", tarArgs, { stdio: "pipe" });

  const size = (statSync(target).size / 1024 / 1024).toFixed(2);
  OK("备份完成", `${target}（${size} MB）`);
  OK("备份时数据量", `工单 ${wo} 条 / 需求 ${req} 条 / 附件 ${att} 个`);
  if (!hasUploads) {
    OK("附件目录", "不存在，本次仅打包数据库");
  }

  // 保留策略：按文件名时间戳排序，删除最旧的。
  // ⚠️ 正则必须**同时认旧版 .db 与新式 .tar.gz**，否则升级后旧备份永不清理。
  const keep = Number(process.env.BACKUP_KEEP || 30);
  const all = readdirSync(backupDir)
    .filter((f) => /^\d{8}-\d{6}\.(db|tar\.gz)$/.test(f))
    .sort();
  if (all.length > keep) {
    const drop = all.slice(0, all.length - keep);
    for (const f of drop) unlinkSync(resolve(backupDir, f));
    OK("清理旧备份", `删除 ${drop.length} 份，保留最近 ${keep} 份`);
  } else {
    OK("保留策略", `现有 ${all.length} 份，上限 ${keep} 份`);
  }

  console.log("\n备份目录：" + backupDir);
} catch (e) {
  FAIL("备份失败", e.message.split("\n")[0]);
  process.exitCode = 1;
} finally {
  await prisma.$disconnect();
  // 无论成败都清掉暂存目录，不留半截快照
  rmSync(stageDir, { recursive: true, force: true });
}
```
